def get_top_n_words(corpus, n=1, k=None):
    """获取 n-gram 特征中词频最高的 Top-K 个词"""
    vec = CountVectorizer(
        ngram_range=(n, n),                     # 设置 n-gram 范围
        stop_words=list(ENGLISH_STOPWORDS)  # 使用停用词
    ).fit(corpus)
    bag_of_words = vec.transform(corpus)        # 生成词频矩阵
    sum_words = bag_of_words.sum(axis=0)
    words_freq = [(word, sum_words[0, idx])
                  for word, idx in vec.vocabulary_.items()]
    words_freq = sorted(words_freq, key=lambda x: x[1], reverse=True)
    return words_freq[:k]

# 获取 3-gram 的 Top-20 高频短语
common_words = get_top_n_words(df['desc'], n=3, k=20)

REPLACE_BY_SPACE = re.compile('[/(){}[\]|@,;]')
BAD_SYMBOLS = re.compile('[^0-9a-z #+_]')

def clean_text(text):
    text = text.lower()()                  # 步骤1: 小写
    text = REPLACE_BY_SPACE.sub(' ', text) # 步骤2: 替换标点
    text = BAD_SYMBOLS.sub('', text)        # 步骤3: 移除非法
    text = ' '.join(
        w for w in text.split()
        if w not in STOPWORDS
    )                                                        # 步骤4: 去停用词
    return text

TF-IDF 原理：

TF（词频）：某个词在文档中出现的次数
IDF（逆文档频率）：计算词的稀有程度，越稀有权重越高
TF-IDF = TF × IDF：突出重要关键词，压制常见词汇

TF-IDF 矩阵形状: (352, N)
特征数量: 数百个（1-gram + 2-gram + 3-gram）
每个酒店 → 一个稀疏的高维向量

余弦相似度含义：

对两个向量 A、B：
cos_sim(A, B) = (A · B) / (|A| × |B|)
取值范围 [0, 1]，值越大表示两个酒店描述越相似。

相似度矩阵: 352 × 352
对角线值 = 1.0（自身完全相似）
非对角线值在 [0, 1) 之间

def recommendations(name, cosine_similarities=cosine_similarities):
    recommended_hotels = []
    idx = indices[indices == name].index[0]  # 找到酒店索引
    score_series = pd.Series(cosine_similarities[idx])
                     .sort_values(ascending=False)  # 降序排列
    top_10_indexes = list(score_series.iloc[1:11].index)  # 排除自身
    for i in top_10_indexes:
        recommended_hotels.append(list(df.index)[i])
    return recommended_hotels
